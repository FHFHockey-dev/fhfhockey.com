/** Every team in pairs, `games` each (BUF and CAR get `shortPair`), with a recent swing that differs by team. */
function league(games: number, window: number, shortPair = games) {
  const scheduled: { id: number; date: string; homeTeamId: number; awayTeamId: number }[] = [], reported: { gameId: number; date: string; team: string }[] = [], appearances: GoalieAppearance[] = [];
  TEAMS.forEach((team, index) => {
    if (index % 2) return;
    const opponent = TEAMS[index + 1], count = team === "BUF" ? shortPair : games;
    for (let g = 0; g < count; g++) {
      const id = (index + 1) * 1000 + g;
      scheduled.push({ id, date: day(g), homeTeamId: teamsInfo[team as "NJD"].id, awayTeamId: teamsInfo[opponent as "NJD"].id });
      for (const side of [team, opponent]) {
        const t = TEAMS.indexOf(side), recent = g >= count - window, threshold = recent ? t % 10 : 3;
        const third = recent && t % 5 === 0 && g === count - 2;
        const slot = third ? 3 : (g * 7 + t * 3) % 10 < threshold ? 2 : 1;
        reported.push({ gameId: id, date: day(g), team: side });
        appearances.push({ goalieId: t * 10 + slot, name: NAMES[(t + (slot - 1) * 3) % NAMES.length], gameId: id, date: day(g), team: side, opponent: null, home: null, started: true, wins: 0, losses: 0, otLosses: 0, saves: 25, shotsAgainst: 28, goalsAgainst: 3, toiSeconds: 3600, shutouts: 0, qualityStart: 0 });
      }
    }
  });
  return { scheduled, reported, appearances, through: day(games) };
}

async function install(page: Page) {
  const user = { id: "00000000-0000-4000-8000-00000000e2e1", aud: "authenticated", role: "authenticated", email: "editor@example.invalid", app_metadata: {}, user_metadata: {} };
  const session = { access_token: "e2e-fictional-editor-token", refresh_token: "e2e-fictional-refresh", token_type: "bearer", expires_in: 365 * 86400, expires_at: Math.floor(Date.now() / 1000) + 365 * 86400, user };
  await page.addInitScript(([key, value]) => window.localStorage.setItem(key, value), [`sb-${supabaseOrigin.hostname.split(".")[0]}-auth-token`, JSON.stringify(session)]);
  await page.route(`${supabaseOrigin.origin}/**`, route => {
    const request = route.request(), url = new URL(request.url());
    if (request.method() !== "GET") return route.abort();
    const one = (row: object) => route.fulfill({ json: /vnd\.pgrst\.object/.test(request.headers().accept ?? "") ? row : [row] });
    if (url.pathname === "/rest/v1/users") return one({ role: "social_editor" });
    if (url.pathname === "/rest/v1/user_profiles") return one({ user_id: user.id, display_name: "Fictional editor", avatar_url: null, timezone: "America/New_York" });
    if (url.pathname === "/rest/v1/user_settings") return one({ user_id: user.id });
    if (url.pathname === "/auth/v1/user") return route.fulfill({ json: user });
    return route.abort();
  });
}
