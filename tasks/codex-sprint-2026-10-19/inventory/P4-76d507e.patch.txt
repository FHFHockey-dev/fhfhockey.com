+const APPROVED_PRESEASON_GAME_IDS = new Set([2026010001, 2026010056]);
+export const APPROVED_PRESEASON_SHIFT_LOOKAHEAD = APPROVED_PRESEASON_GAME_IDS.size;
+
+export type PreseasonShiftSkip = {
+  code: "APPROVED_PRESEASON_SHIFT_SKIP";
+  gameId: number;
+  gameType: 1;
+  stage: string;
+  reason: string;
+  outcome: "skipped";
+  inputCompleteness: "unverified";
+  eligibleForDerivedOrAccuracy: false;
+};
+
+/** Only the two operator-approved games and known input defects qualify. */
+export function approvedPreseasonShiftSkip(args: {
+  gameId: number;
+  gameType: unknown;
+  stage: string;
+  error: unknown;
+  retry?: boolean;
+}): PreseasonShiftSkip | null {
+  if (
+    args.retry ||
+    args.gameType !== 1 ||
+    !APPROVED_PRESEASON_GAME_IDS.has(args.gameId)
+  ) return null;
+
+  const reason = args.error instanceof Error ? args.error.message :
+    args.error && typeof args.error === "object" && "message" in args.error
+      ? String(args.error.message) : String(args.error);
+  const allowed =
+    (args.stage === "capture_raw_sources" &&
+      reason === "INVALID_PROJECTION_RAW_SHIFT_ROWS") ||
+    (args.stage === "relationship_roster" &&
+      reason === `Invalid relationship roster scope for game ${args.gameId}`);
+  if (!allowed) return null;
+
+  return {
+    code: "APPROVED_PRESEASON_SHIFT_SKIP",
+    gameId: args.gameId,
+    gameType: 1,
+    stage: args.stage,
+    reason,
+    outcome: "skipped",
+    inputCompleteness: "unverified",
+    eligibleForDerivedOrAccuracy: false,
+  };
+}
